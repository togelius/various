=====
RULES
=====

[ > player | void ] -> [ player | void ]
horizontal [ > Player | VBeam ] -> [ > Player | > VBeam ]
vertical [ Wall | > Player | HBeam ] -> [ Wall | > Player | > HBeam ]
[ Whale | ... | perpendicular Beam | Wall ] -> [ perpendicular Whale | ... | perpendicular Beam | Wall ]
[ Player | ... | perpendicular Beam | Wall ] -> [ perpendicular Player | ... | perpendicular Beam | Wall ]
[ perpendicular Player | ... | perpendicular Beam | Wall ] -> [ Player | ... | perpendicular Beam | Wall ]
[ Whale Void ] -> [ Void ]
right [ Beam ] -> [  ]
late horizontal [ Whale | ... |  ] -> [ Whale | ... | HBeam ]
random [ HBeam | no HBeam no Whale ] -> [ HBeam | HBeam ]
[ > Player | Whale ] -> [ > Player |  ]
vertical [ Whale | ... |  ] -> [ Whale | ... | VBeam ]
